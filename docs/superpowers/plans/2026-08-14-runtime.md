# Runtime Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the levels playable — a side-scrolling platformer runtime inside `index.html`, paced like Super Mario Bros.

**Architecture:** One fenced `/* ===== RUNTIME ===== */` section in `index.html`, split into six modules with inward-pointing dependencies (`compileLevel`, `physics`, `actors`, `runLoop`, `runRender`, `runInput`). Simulation is pure functions over a `RunState` object; rendering and input sit outside them. The editor's item model is compiled once into a flat solidity grid rather than queried per frame. Editor state is read at compile time and never written.

**Tech Stack:** Vanilla ES2020, canvas 2D, no dependencies, no build step. Tests run in-browser via a `?test=1` mode in the same file.

**Spec:** `docs/superpowers/specs/2026-08-13-runtime-design.md`

## Global Constraints

- **C-1** Single file. All code goes in `index.html`. No asset files, no CDN fetches, no build step.
- **C-2** All artwork procedural. Reuse sprites generated at boot; draw no new art.
- **C-3** No browser storage. `localStorage` is prohibited.
- **C-4** Items are immutable. The runtime must never mutate `track`, `project`, or any editor item in place. Compile and copy. Violating this corrupts undo history and duplicated levels.
- **R-15** Target ~100 lines per function. **Any function over 300 lines is a blocking defect.**
- **R-13** Dependencies point inward. `physics` and `runInput` depend on nothing. Nothing depends on `runRender`. No circular dependencies.
- **R-14** Physics integration, wake rules and contact outcomes are pure functions of `(state, input)` — no canvas, DOM or event access inside them.
- Constants are exact. Copy them verbatim from the spec's §4 table; do not round or re-tune during implementation.

/* ===================================================================
   SELF TESTS  —  index.html?test=1
   No framework, no build step (C-1). Assertions run in-page and report
   into a <pre>. Simulation is pure (R-14), so nearly everything below
   is testable without rendering a frame.
   =================================================================== */
const T=(()=>{
  const lines=[]; let passed=0, failed=0, group='';
  const log=(ok,label)=>{
    ok?passed++:failed++;
    lines.push(`${ok?'  ok':'FAIL'}  ${group}${group?' › ':''}${label}`);
  };
  return {
    group(name,fn){ group=name; try{ fn(); }catch(e){ log(false,`threw: ${e.message}`); } group=''; },
    ok(c,label){ log(!!c,label); },
    eq(a,b,label){ log(Object.is(a,b),`${label} (got ${a}, want ${b})`); },
    close(a,b,tol,label){ log(Math.abs(a-b)<=tol,`${label} (got ${a}, want ${b}±${tol})`); },
    run(){ return {passed,failed,lines}; }
  };
})();
function runSelfTests(){
  SELF_TESTS.forEach(fn=>fn());
  const {passed,failed,lines}=T.run();
  const pre=document.createElement('pre');
  pre.id='testout';
  pre.style.cssText='position:fixed;inset:0;z-index:99999;margin:0;padding:16px;'+
    'overflow:auto;background:#1a1a1a;color:#dcdcdc;font:12px/1.5 ui-monospace,monospace';
  pre.textContent=`${failed?'FAILED':'PASSED'}  ${passed} passed, ${failed} failed\n\n`+lines.join('\n');
  document.body.appendChild(pre);
  window.__testResult={passed,failed};
}
const SELF_TESTS=[];
```

const PHY={
  WALK_ACCEL:0.10, RUN_ACCEL:0.16, MAX_WALK:1.5, MAX_RUN:2.5,
  FRICTION:0.12,   SKID_DECEL:0.25, AIR_CONTROL:0.08,
  GRAVITY:0.45,    GRAVITY_HELD:0.22,
  JUMP_IMPULSE:-4.0, JUMP_IMPULSE_RUN:-5.0, MAX_FALL:4.5,
  COYOTE_TICKS:4,  JUMP_BUFFER_TICKS:6,
  ENEMY_WALK:0.5,  STOMP_BOUNCE:-3.0,
  BOY_W:10, BOY_H:24, VIEW_COLS:16,
  DIE_FRAMES:45, WIN_FRAMES:120, MAX_CATCHUP:5, MS_PER_TICK:1000/60
};

/* Level edges are solid so the boy cannot leave the track. Above the
   level is open air; below it is open, which is what makes pits fatal. */
function solidAt(grid,c,r){
  if(c<0||c>=COLS) return 1;
  if(r<0||r>=ROWS) return 0;
  return grid[r*COLS+c];
}

function compileLevel(lv){
  const grid=new Uint8Array(COLS*ROWS);
  const mark=(c,r)=>{ if(c>=0&&c<COLS&&r>=0&&r<ROWS) grid[r*COLS+c]=1; };
  let flag=null;
  for(const it of lv.fg){
    if(it.t==='shape'){
      for(const k of it.cells.keys()){
        const [dx,dy]=k.split(',').map(Number);
        mark(it.ax+dx,it.ay+dy);                        /* R-8: every material */
      }
    } else if(it.t==='stamp'){
      if(it.id==='CASTLE'){                             /* R-9: castle solid */
        const c0=it.ax-sAncCol(it.id), r0=it.ay-sAncRow(it.id);
        for(let r=0;r<sH(it.id);r++) for(let c=0;c<sW(it.id);c++) mark(c0+c,r0+r);
      } else if(it.id==='FLAG'){
        flag={c0:it.ax-sAncCol(it.id),r0:it.ay-sAncRow(it.id),
              w:sW(it.id),h:sH(it.id)};                 /* R-9: trigger */
      }
      /* CLOUD falls through: scenery */
    }
  }
  const boyRec=lv.actors.find(a=>a.id==='PLAYER');
  if(!boyRec) return {error:'no boy'};                  /* E-1 */
  if(!flag)   return {error:'no flagpole'};             /* E-2 */

  const spawn=freeSpawn(grid,boyRec.ax*TILE+(TILE-PHY.BOY_W)/2,
                             (boyRec.ay+1)*TILE-PHY.BOY_H);
  if(!spawn) return {error:'the boy is buried — move him somewhere open'}; /* E-3 */

  const actors=lv.actors.filter(a=>a.id!=='PLAYER').map(a=>{
    const art=ACTORS[a.id];
    return {id:a.id, w:art.img.width, h:art.img.height,
      x:a.ax*TILE, y:(a.ay+1)*TILE-art.img.height,
      x0:a.ax*TILE, y0:(a.ay+1)*TILE-art.img.height,   /* R-19 snapshot */
      dir:a.dir||-1, dir0:a.dir||-1, vy:0,
      ledge:art.ledge, awake:false, alive:true, onGround:false};
  });
  return {grid,flag,actors,spawn,
    boy:newBody(spawn.x,spawn.y),
    camX:0, phase:'playing', timer:0, dir:1};
}

function newBody(x,y){
  return {x,y,vx:0,vy:0,w:PHY.BOY_W,h:PHY.BOY_H,
    onGround:false,coyote:0,buffer:0};
}

/* E-3: if the boy was placed inside geometry, lift him to the nearest
   clear space above rather than starting him embedded in a wall. */
function freeSpawn(grid,x,y){
  for(let lift=0;lift<ROWS;lift++){
    const ty=y-lift*TILE;
    if(!boxHitsSolid(grid,x,ty,PHY.BOY_W,PHY.BOY_H)) return {x,y:ty};
  }
  return null;
}
function boxHitsSolid(grid,x,y,w,h){
  const c0=Math.floor(x/TILE), c1=Math.floor((x+w-1)/TILE);
  const r0=Math.floor(y/TILE), r1=Math.floor((y+h-1)/TILE);
  for(let r=r0;r<=r1;r++) for(let c=c0;c<=c1;c++) if(solidAt(grid,c,r)) return true;
  return false;
}

/* R-10: one axis at a time, horizontal then vertical. Resolving both at
   once snags on tile corners. Max speed (4.5) is under TILE (16), so a
   single step can never tunnel. */
function moveX(box,dx,grid){
  if(!dx) return;
  box.x+=dx;
  const r0=Math.floor(box.y/TILE), r1=Math.floor((box.y+box.h-1)/TILE);
  if(dx>0){
    const c=Math.floor((box.x+box.w-1)/TILE);
    for(let r=r0;r<=r1;r++) if(solidAt(grid,c,r)){ box.x=c*TILE-box.w; box.vx=0; box.hitWall=true; return; }
  } else {
    const c=Math.floor(box.x/TILE);
    for(let r=r0;r<=r1;r++) if(solidAt(grid,c,r)){ box.x=(c+1)*TILE; box.vx=0; box.hitWall=true; return; }
  }
}
function moveY(box,dy,grid){
  box.onGround=false;
  if(!dy) return;
  box.y+=dy;
  const c0=Math.floor(box.x/TILE), c1=Math.floor((box.x+box.w-1)/TILE);
  if(dy>0){
    const r=Math.floor((box.y+box.h-1)/TILE);
    for(let c=c0;c<=c1;c++) if(solidAt(grid,c,r)){ box.y=r*TILE-box.h; box.vy=0; box.onGround=true; return; }
  } else {
    const r=Math.floor(box.y/TILE);
    for(let c=c0;c<=c1;c++) if(solidAt(grid,c,r)){ box.y=(r+1)*TILE; box.vy=0; return; }
  }
}

/* R-14: pure over (body, intent, grid). No DOM, no canvas, no events. */
function stepBoy(b,intent,grid){
  const maxSpd=intent.run?PHY.MAX_RUN:PHY.MAX_WALK;
  const accel=b.onGround?(intent.run?PHY.RUN_ACCEL:PHY.WALK_ACCEL):PHY.AIR_CONTROL;
  if(intent.left)       b.vx-=(b.vx>0?PHY.SKID_DECEL:accel);
  else if(intent.right) b.vx+=(b.vx<0?PHY.SKID_DECEL:accel);
  else if(b.onGround)   b.vx-=Math.sign(b.vx)*Math.min(Math.abs(b.vx),PHY.FRICTION);
  b.vx=Math.max(-maxSpd,Math.min(maxSpd,b.vx));
  if(b.vx) b.facing=Math.sign(b.vx);

  /* R-4: buffer an early press, and allow a late one after a ledge. */
  b.buffer=intent.jump?PHY.JUMP_BUFFER_TICKS:Math.max(0,b.buffer-1);
  b.coyote=b.onGround?PHY.COYOTE_TICKS:Math.max(0,b.coyote-1);
  if(b.buffer>0&&b.coyote>0){
    /* R-3: impulse scales with horizontal speed. */
    b.vy=Math.abs(b.vx)>=PHY.MAX_WALK?PHY.JUMP_IMPULSE_RUN:PHY.JUMP_IMPULSE;
    b.buffer=0; b.coyote=0; b.onGround=false;
  }
  /* R-2: lighter gravity while rising with the button held. */
  const g=(intent.jumpHeld&&b.vy<0)?PHY.GRAVITY_HELD:PHY.GRAVITY;
  b.vy=Math.min(PHY.MAX_FALL,b.vy+g);

  moveX(b,b.vx,grid);
  moveY(b,b.vy,grid);
}
```

/* R-21: wake on entering the view plus a one-tile margin, so nothing is
   seen popping into motion at the screen edge.
   R-22: there is no path back to dormant. Once awake, awake for the run —
   "everything you made stays there". Only respawn resets this. */
function wakeCheck(st){
  const L=st.camX-TILE, R=st.camX+PHY.VIEW_COLS*TILE+TILE;
  for(const a of st.actors)
    if(!a.awake && a.x+a.w>L && a.x<R) a.awake=true;
}

/* R-17: behaviour is driven by the `ledge` property the editor already
   records on each ACTORS entry. Adding an enemy means adding a table
   entry, not editing a switch in here. */
function stepActor(a,grid){
  if(!a.awake||!a.alive) return;
  a.vy=Math.min(PHY.MAX_FALL,a.vy+PHY.GRAVITY);
  moveY(a,a.vy,grid);
  if(a.ledge!=='none'){
    if(a.onGround&&a.ledge==='turn'){
      const ahead=Math.floor((a.dir>0?a.x+a.w+PHY.ENEMY_WALK:a.x-PHY.ENEMY_WALK)/TILE);
      const below=Math.floor((a.y+a.h)/TILE);
      if(!solidAt(grid,ahead,below)) a.dir*=-1;
    }
    a.hitWall=false;
    moveX(a,PHY.ENEMY_WALK*a.dir,grid);
    if(a.hitWall) a.dir*=-1;
  }
  if(a.y>ROWS*TILE) a.alive=false;                     /* R-23 */
}

function renderRun(st){

  buf.x.clearRect(0,0,buf.c.width,buf.c.height);

let runState=null, runRAF=0, runAcc=0, runLast=0;
function runFrame(t){
  if(!runState) return;
  if(!runLast) runLast=t;
  runAcc+=Math.min(250,t-runLast); runLast=t;           /* E-12 clamp */
